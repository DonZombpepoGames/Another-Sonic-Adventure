event ObjectUpdate
	BoxCollisionTest(C_TOUCH, object.entityPos, 0, 0, 32, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
	if checkResult == true
		CreateTempObject(TypeName[Title Card], 0, object.xpos, object.ypos)
		object[tempObjectPos].drawOrder = 7
		object[tempObjectPos].priority = PRIORITY_ACTIVE
		ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
	end if
end event
