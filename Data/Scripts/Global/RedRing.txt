	2, 0
	3, 0
	4, 0
	3, 1
	2, 1
	1, 1
end table

// ========================
// Events
// ========================

event ObjectUpdate
	object.animationTimer++
	if object.animationTimer == 8
		object.animationTimer = 0
		GetTableValue(object.frame, object.frameCount, RedRing_FrameTable)
		object.frameCount++
		GetTableValue(object.direction, object.frameCount, RedRing_FrameTable)
		object.frameCount++
		object.frameCount &= 15
	end if

	// Check interaction with players
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)

		// Make sure that the player isn't supposed to ignore rings right now
		CheckEqual(player[0].state, Player_State_Hurt)
		temp0 = checkResult
		CheckEqual(player[currentPlayer].state, Player_State_Hurt)
		temp0 = checkResult
		CheckEqual(player[currentPlayer].state, Player_State_GotHit)
		temp0 |= checkResult

		if temp0 == false
			// All checks passed, see if the player is touching the ring

			BoxCollisionTest(C_TOUCH, object.entityPos, -14, -14, 14, 14, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				// Player is touching ring - collect it

				// Turn this object into a sparkle
				object.type = TypeName[Blank Object]
				PlaySfx(SfxName[Red Ring], false)
				// Set this object to the same drawOrder as the player
				object.drawOrder = player[currentPlayer].currentPlane

				// Add it to the player's red ring count
				coinCount++		
			end if
		end if
	next
end event

event ObjectDraw
	DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
end event

	// Add the Ring to the debug mode object list	
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
			checkResult = object.propertyValue
			break
			
		case 0 // type
			checkResult = object.propertyValue
			break
			
		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
			object.propertyValue = editor.variableValue
			break
			
		case 0 // type
			object.propertyValue = editor.variableValue
			break
			
		end switch
	end if
end event


event RSDKDraw

end event
